type qualificationCountingReader struct {
	reads int
}

func (r *qualificationCountingReader) Read([]byte) (int, error) {
	r.reads++
	return 0, errors.New("unexpected underlying read")
}

func qualificationLimits() SuccessorIngressLimits {
	return SuccessorIngressLimits{
		HeaderBytes:             1 << 20,
		FrameBytes:              1 << 20,
		ConsumptionBytes:        1 << 24,
		AcquisitionBytes:        1 << 24,
		MaxReadBytes:            4096,
		PrefetchBytes:           4096,
		HistoryAcquiredBytes:    1 << 24,
		HistoryOutstandingBytes: 1 << 24,
	}
}

type qualificationScriptedRead struct {
	data     []byte
	err      error
	requests *[]int
}

type qualificationScriptedReader struct {
	reads    []qualificationScriptedRead
	requests []int
}

func (r *qualificationScriptedReader) Read(p []byte) (int, error) {
	r.requests = append(r.requests, len(p))
	if len(r.reads) == 0 {
		return 0, io.EOF
	}
	next := r.reads[0]
	r.reads = r.reads[1:]
	if next.requests != nil {
		*next.requests = append(*next.requests, len(p))
	}
	n := copy(p, next.data)
	return n, next.err
}

type qualificationCapture struct {
	frames [][]byte
}

func (c *qualificationCapture) ObserveOriginalFrame(frame []byte) {
	c.frames = append(c.frames, append([]byte(nil), frame...))
}

func qualificationFrame(body []byte) []byte {
	return append([]byte(fmt.Sprintf("Content-Length: %d\r\n\r\n", len(body))), body...)
}

func qualificationReader(t *testing.T, limits SuccessorIngressLimits) *SuccessorIngressReader {
	t.Helper()
	r, err := NewSuccessorIngressReader(bytes.NewReader(nil), SuccessorIngressOptions{Limits: limits, OwnerID: "prospective-q1"})
	if err != nil {
		t.Fatalf("setup: NewSuccessorIngressReader: %v", err)
	}
	return r
}

func TestProspectiveQualification(t *testing.T) {
	t.Run("Q1-B01", func(t *testing.T) {
		// Setup: synthetic C=8388607 with two unread prefetch bytes.
		limits := qualificationLimits()
		limits.ConsumptionBytes = 8388608
		r := qualificationReader(t, limits)
		r.consumed = 8388607
		r.prefetch = append(r.prefetch, 'x', 'x')
		r.recorder = &ingressTestRecorder{limit: 8}

		got := r.step(ingressStepInput{Checkpoint: ingressConsume, Consume: 2})
		if got.Consumed != 2 || r.consumed != 8388609 || !r.sealed || got.OutcomeKind != ingressOutcomeConsumptionLimit || !errors.Is(got.Outcome, errIngressConsume) {
			t.Fatalf("GOOD consume boundary: result=%+v consumed=%d sealed=%v", got, r.consumed, r.sealed)
		}
		if len(r.recorder.events) != 3 || r.recorder.events[0].Stage != ingressEventConsume || r.recorder.events[1].Stage != ingressEventSeal || r.recorder.events[2].Stage != ingressEventOutcome {
			t.Fatalf("GOOD event sequence consume/seal/outcome: %+v", r.recorder.events)
		}
	})

	t.Run("Q1-B02", func(t *testing.T) {
		// Setup: synthetic A equals the acquisition cap before pre-read.
		limits := qualificationLimits()
		limits.AcquisitionBytes = 8392705
		transport := &qualificationCountingReader{}
		r, err := NewSuccessorIngressReader(transport, SuccessorIngressOptions{Limits: limits, OwnerID: "prospective-q1"})
		if err != nil {
			t.Fatalf("setup: NewSuccessorIngressReader: %v", err)
		}
		r.acquired = 8392705
		beforeC, beforeV := r.consumed, r.validated

		got := r.step(ingressStepInput{Checkpoint: ingressPreRead})
		if got.OutcomeKind != ingressOutcomeAcquisitionLimit || !errors.Is(got.Outcome, errIngressAcquire) || got.NeedRead != 0 || !r.sealed || r.terminalOutcome != ingressOutcomeAcquisitionLimit {
			t.Fatalf("GOOD exact acquisition refusal: result=%+v sealed=%v terminal=%v", got, r.sealed, r.terminalOutcome)
		}
		if transport.reads != 0 || r.acquired != 8392705 || r.consumed != beforeC || r.validated != beforeV {
			t.Fatalf("GOOD zero-read immutable A/C/V: reads=%d A/C/V=%d/%d/%d", transport.reads, r.acquired, r.consumed, r.validated)
		}
	})

	t.Run("Q1-B03", func(t *testing.T) {
		// Setup: fixed prefetch backing is full; all other limits are nonbinding.
		r := qualificationReader(t, qualificationLimits())
		r.prefetch = append(r.prefetch, make([]byte, 4096)...)
		got := r.step(ingressStepInput{Checkpoint: ingressPreRead})
		if got.NeedRead != 0 || len(r.prefetch) != 4096 || cap(r.prefetch) != 4096 || r.storage.live != 4096 || len(r.storage.records) != 1 {
			t.Fatalf("GOOD full fixed backing: NeedRead=%d len/cap=%d/%d live=%d records=%d", got.NeedRead, len(r.prefetch), cap(r.prefetch), r.storage.live, len(r.storage.records))
		}
	})

	t.Run("Q1-B04", func(t *testing.T) {

		r := qualificationReader(t, qualificationLimits())
		r.prefetch = append(r.prefetch, make([]byte, 5)...)
		got := r.step(ingressStepInput{Checkpoint: ingressPreRead})

		}
