func privateCallHierarchyProfileDigestV1(id, domain string) [sha256.Size]byte {
	preimage := make([]byte, 0, len(privateCallsLSPProfileDigestConstructorDomainV1)+8+len(domain)+len(id))
	preimage = append(preimage, privateCallsLSPProfileDigestConstructorDomainV1...)
	var length [4]byte
	binary.BigEndian.PutUint32(length[:], uint32(len(domain)))
	preimage = append(preimage, length[:]...)
	preimage = append(preimage, domain...)
	binary.BigEndian.PutUint32(length[:], uint32(len(id)))
	preimage = append(preimage, length[:]...)
	preimage = append(preimage, id...)
	return sha256.Sum256(preimage)
}

func compilePlannedCallSitesForDirection(profile privateCallHierarchyProfileIdentityV1, direction privateCallHierarchyDirectionV1) {
	m := &Manager{directional: newPrivateCallHierarchyLedgerV1()}
	owner := &callHierarchyLayer1GenericOwner{}
	lease := callHierarchyLayer1GenericLease{owner: owner}
	target := privateCallHierarchyTargetIdentityV1{Ordinal: 1}

	r, stage, refusal := m.reserveCallHierarchyV1(lease, profile, direction, target, privateCallHierarchyDirectionalMeasurementMaxBytesV1)
	_, _, _ = m.reserveCallHierarchyV1(lease, profile, direction, target, privateCallHierarchyDirectionalMeasurementMaxBytesV1+1)
	_, _, _ = m.reserveCallHierarchyV1(lease, privateCallHierarchyProfileIdentityV1{}, privateCallHierarchyDirectionInvalidV1, privateCallHierarchyTargetIdentityV1{}, 0)
	var nilManager *Manager
	_, _, _ = nilManager.reserveCallHierarchyV1(callHierarchyLayer1GenericLease{}, privateCallHierarchyProfileIdentityV1{}, privateCallHierarchyDirectionInvalidV1, privateCallHierarchyTargetIdentityV1{}, 0)
	if r == nil {
		r = &privateCallHierarchyReservationV1{}
	}

	charge := privateB4ByteLeaseV2{}
	custody, stage, refusal := m.transferCallHierarchyComponentV1(lease, r, privateCallHierarchyComponentPrepareResponseV1, 1, 2, []byte("[]"), &charge, privateCallHierarchyLifetimeBorrowedV1)
	_, _, _ = m.transferCallHierarchyComponentV1(lease, r, privateCallHierarchyComponentInvalidV1, 0, 0, nil, nil, privateCallHierarchyLifetimeInvalidV1)
	stage, refusal = m.advanceCallHierarchyV1(lease, r, privateCallHierarchyStateReservedV1, privateCallHierarchyStatePreparedRawOwnedV1)
	a, err := scanCallHierarchyRawArrayV1([]byte("[{}]"), 1)
	_, _ = scanCallHierarchyRawArrayV1(nil, 0)
	s, err := selectCallHierarchyPreparedOrdinalV1(a, 0)
	_, _ = selectCallHierarchyPreparedOrdinalV1(privateCallHierarchyRawArrayV1{}, 1)
	_, err = assembleIncomingCallHierarchyParamsV1(s)
	_, err = assembleOutgoingCallHierarchyParamsV1(s)
	_, stage, refusal, err = m.handoffCallHierarchySelectedRawV1(lease, r, custody, s)
	_, _, _, _ = m.handoffCallHierarchySelectedRawV1(callHierarchyLayer1GenericLease{}, nil, privateCallHierarchyCustodyV1{}, privateCallHierarchyRawSelectionV1{})
	_, _ = m.viewCallHierarchyV1(lease, r)
	_, _ = m.viewCallHierarchyV1(callHierarchyLayer1GenericLease{}, nil)

	dispositions := [...]privateCallHierarchyDispositionV1{
		privateCallHierarchyDispositionInvalidV1, privateCallHierarchyDispositionCompleteV1,
		privateCallHierarchyDispositionCompleteEmptyNullV1, privateCallHierarchyDispositionCompleteEmptyArrayV1,
		privateCallHierarchyDispositionFailedV1, privateCallHierarchyDispositionUnsupportedV1,
		privateCallHierarchyDispositionCancelledV1, privateCallHierarchyDispositionTimedOutV1,
		privateCallHierarchyDispositionRefusedV1, privateCallHierarchyDispositionPartialV1,
	}
	terminals := [...]privateCallHierarchyTerminalV1{
		privateCallHierarchyTerminalNoneV1, privateCallHierarchyTerminalCompleteV1,
		privateCallHierarchyTerminalCompleteEmptyNullV1, privateCallHierarchyTerminalCompleteEmptyArrayV1,
		privateCallHierarchyTerminalTransportErrorV1, privateCallHierarchyTerminalProtocolErrorV1,
		privateCallHierarchyTerminalMissingPayloadV1, privateCallHierarchyTerminalMalformedV1,
		privateCallHierarchyTerminalPartialCaptureV1, privateCallHierarchyTerminalLimitRefusedV1,
		privateCallHierarchyTerminalPolicyRefusedV1, privateCallHierarchyTerminalUnsupportedV1,
		privateCallHierarchyTerminalCancelledV1, privateCallHierarchyTerminalTimedOutV1,
	}
	refusals := [...]privateCallHierarchyRefusalV1{
		privateCallHierarchyRefusalNoneV1, privateCallHierarchyRefusalInvalidV1, privateCallHierarchyRefusalNilV1,
		privateCallHierarchyRefusalForeignV1, privateCallHierarchyRefusalStaleV1, privateCallHierarchyRefusalSlotBusyV1,
		privateCallHierarchyRefusalLimitV1, privateCallHierarchyRefusalOverflowV1, privateCallHierarchyRefusalStateV1,
		privateCallHierarchyRefusalMutationV1, privateCallHierarchyRefusalIdentityV1,
		privateCallHierarchyRefusalMalformedV1, privateCallHierarchyRefusalTerminalRaceV1,
	}
	lifetimes := [...]privateCallHierarchyLifetimeV1{
		privateCallHierarchyLifetimeInvalidV1, privateCallHierarchyLifetimeBorrowedV1,
		privateCallHierarchyLifetimeCopiedV1, privateCallHierarchyLifetimeAliasedV1,
		privateCallHierarchyLifetimeViewV1, privateCallHierarchyLifetimeReleasedV1,
	}
	accounting := privateCallHierarchyTargetAccountingV1{
		DenominatorKnown: true,
		Denominator:      3,
		Attempted:        2,
		Unattempted:      1,
		CompleteSuccess:  1,
		Failed:           1,
		AttemptOutcomes: []privateCallHierarchyAttemptOutcomeV1{
			{TargetOrdinal: 0, Disposition: privateCallHierarchyDispositionCompleteV1, Terminal: privateCallHierarchyTerminalCompleteV1},
			{TargetOrdinal: 1, Disposition: privateCallHierarchyDispositionFailedV1, Terminal: privateCallHierarchyTerminalTransportErrorV1},
		},
		Frontier: privateCallHierarchyFrontierV1{
			Present: true, TargetOrdinal: 2, Stage: privateCallHierarchyStageDirectionalCustodyV1,
			Reason: privateCallHierarchyRefusalLimitV1, UnattemptedOrdinals: []uint64{2},
		},
	}
	for i, disposition := range dispositions {
		terminal := terminals[i%len(terminals)]
		refusal = refusals[i%len(refusals)]
		_ = m.terminalCallHierarchyV1(lease, r, privateCallHierarchyStageTerminalV1, disposition, terminal, refusal, accounting)
	}
	for _, terminal := range terminals {
		_ = m.terminalCallHierarchyV1(lease, r, privateCallHierarchyStageTerminalV1, privateCallHierarchyDispositionFailedV1, terminal, privateCallHierarchyRefusalNoneV1, accounting)
	}
	for _, refusal = range refusals {
		_ = m.terminalCallHierarchyV1(lease, r, privateCallHierarchyStageDirectionalCustodyV1, privateCallHierarchyDispositionRefusedV1, privateCallHierarchyTerminalLimitRefusedV1, refusal, accounting)
	}
	for _, lifetime := range lifetimes {
		_, _, _ = m.transferCallHierarchyComponentV1(lease, r, privateCallHierarchyComponentObservationV1, 2, 2, []byte("{}"), &charge, lifetime)
	}

	owner.directionalLinkV1 = privateCallHierarchyTerminalLinkV1{manager: m, reservation: r}
	lease.directionalV1 = &owner.directionalLinkV1
	_ = lease.Release()
	_ = lease.Release()
	owner.requestTerminal()
	owner.requestTerminal()
	_ = m.releaseCallHierarchyV1(lease, r)
	_ = m.releaseCallHierarchyV1(lease, r)
	_ = m.releaseCallHierarchyTerminalV1(owner, r)
	_ = m.releaseCallHierarchyTerminalV1(owner, r)
	_, _, _, _ = stage, refusal, custody, err
}

func compileAllPlannedCallSites() {
	compilePlannedCallSitesForDirection(newPrivateCallsLSPIncomingProfileV1(), privateCallHierarchyDirectionIncomingV1)
	compilePlannedCallSitesForDirection(newPrivateCallsLSPOutgoingProfileV1(), privateCallHierarchyDirectionOutgoingV1)
}
