func privateCallHierarchyProfileDigestV3(id, domain string) [sha256.Size]byte {
	p := []byte(privateCallsLSPProfileDigestDomainV1)
	var n [4]byte
	binary.BigEndian.PutUint32(n[:], uint32(len(domain)))
	p = append(p, n[:]...)
	p = append(p, domain...)
	binary.BigEndian.PutUint32(n[:], uint32(len(id)))
	p = append(p, n[:]...)
	p = append(p, id...)
	return sha256.Sum256(p)
}
