func compileAllPlannedCallSites() {
	m := &Manager{directional: newPrivateCallHierarchyDirectionalLedgerV1()}
	owner := &callHierarchyLayer1GenericOwner{}
	lease := callHierarchyLayer1GenericLease{owner: owner}
	r, stage, refusal := m.reserveCallHierarchyDirectionalV1(lease, privateCallHierarchyDirectionIncomingV1, privateCallHierarchyDirectionalMeasurementMaxBytesV1)
	_, _ = stage, refusal
	if r == nil {
		r = &privateCallHierarchyDirectionalReservationV1{}
	}
	charge := privateB4ByteLeaseV2{}
	custody, stage, refusal := m.transferCallHierarchyDirectionalComponentV1(lease, r, privateCallHierarchyDirectionalComponentPrepareResponseV1, 1, 2, []byte("[]"), &charge, privateCallHierarchySourceFormRawResultV1, privateCallHierarchyDirectionalOwnershipTransferredV1)
	_, _ = stage, refusal
	stage, refusal = m.advanceCallHierarchyDirectionalV1(lease, r, privateCallHierarchyDirectionalStateReservedV1, privateCallHierarchyDirectionalStatePreparedRawOwnedV1)
	a, err := scanCallHierarchyRawArrayV1([]byte("[{}]"), 1)
	s, err := selectCallHierarchyPreparedOrdinalV1(a, 0)
	_, err = assembleIncomingCallHierarchyParamsV1(s)
	_, err = assembleOutgoingCallHierarchyParamsV1(s)
	_, stage, refusal, err = m.handoffCallHierarchySelectedRawV1(lease, r, custody, s)
	_, _ = m.viewCallHierarchyDirectionalV1(lease, r)
	stage, refusal = m.observeCallHierarchyDirectionalV1(lease, r, privateCallHierarchyDirectionalTerminalCompleteV1)
	_ = m.refuseCallHierarchyDirectionalV1(lease, r, privateCallHierarchyDirectionalRefusalStageObservationV1, privateCallHierarchyDirectionalRefusalTerminalRaceV1, privateCallHierarchyDirectionalTerminalPartialCaptureV1)
	link := privateCallHierarchyDirectionalTerminalLinkV1{manager: m, reservation: r}
	owner.directionalLinkV1 = link
	lease.directionalV1 = &owner.directionalLinkV1
	_ = lease.Release()
	owner.requestTerminal()
	_ = m.releaseCallHierarchyDirectionalV1(lease, r)
	_ = m.releaseCallHierarchyDirectionalTerminalV1(owner, r)
	_, _, _, _ = stage, refusal, custody, err
}
